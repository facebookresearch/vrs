38template <typename T>
39class DataPieceVector : public DataPiece {
40 public:
42 explicit DataPieceVector(string label, vector<T> defaultValues = {})
43 : DataPiece(std::move(label), DataPieceType::Vector, DataLayout::kVariableSize),
44 defaultValues_{std::move(defaultValues)} {}
47 explicit DataPieceVector(const MakerBundle& bundle);
48
51 const string& getElementTypeName() const override {
52 return vrs::getTypeName<T>();
53 }
56 size_t getVariableSize() const override {
57 return sizeof(T) * stagedValues_.size();
58 }
63 size_t collectVariableData(int8_t* data, size_t bufferSize) override {
64 const size_t writtenSize = min(bufferSize, getVariableSize());
65 if (writtenSize > 0) {
66 unalignedCopy(data, stagedValues_.data(), writtenSize);
67 }
68 return writtenSize;
69 }
70
73 const vector<T>& stagedValues() const {
74 return stagedValues_;
75 }
78 vector<T>& stagedValues() {
79 return stagedValues_;
80 }
83 void stage(const vector<T>& values) {
84 stagedValues_ = values;
85 }
88 void stage(vector<T>&& values) {
89 stagedValues_ = std::move(values);
90 }
91
95 void stage(const T* values, size_t count) {
96 stagedValues_.resize(count);
97 if (count > 0) {
98 unalignedCopy(stagedValues_.data(), values, count * sizeof(T));
99 }
100 }
103 template <size_t n>
104 void stage(const T (&arr)[n]) {
105 stage(arr, n);
106 }
107
111 bool get(vector<T>& outValues) const {
112 size_t count = 0;
113 const T* ptr = layout_.getVarData<T>(offset_, count);
114 if (ptr != nullptr && count > 0) {
115 outValues.resize(count);
116 unalignedCopy(outValues.data(), ptr, count * sizeof(T));
117 return true;
118 }
119 if (pieceIndex_ != DataLayout::kNotFound) {
120 outValues.clear();
121 return true;
122 }
123 outValues = defaultValues_;
124 return false;
125 }
132 template <
133 typename U = T,
134 std::enable_if_t<std::is_same<U, T>::value && std::is_trivially_copyable<U>::value, int> = 0>
135 bool getRawData(const void*& outData, size_t& outCount) const {
136 outData = layout_.getVarData<T>(offset_, outCount);
137 if (outData != nullptr && outCount > 0) {
138 return true;
139 }
140 outData = nullptr;
141 outCount = 0;
142 return pieceIndex_ != DataLayout::kNotFound;
143 }
144
150 void setDefault(const T* defaultValues, size_t count) {
151 defaultValues_.resize(count);
152 if (count > 0) {
153 unalignedCopy(defaultValues_.data(), defaultValues, count * sizeof(T));
154 }
155 }
160 template <size_t n>
161 void setDefault(const T (&arr)[n]) {
162 setDefault(arr, n);
163 }
168 void setDefault(const vector<T>& values) {
169 defaultValues_ = values;
170 }
175 void setDefault(vector<T>&& values) {
176 defaultValues_ = std::move(values);
177 }
180 const vector<T>& getDefault() const {
181 return defaultValues_;
182 }
184 void initToDefault() override {
185 stagedValues_ = defaultValues_;
186 }
187
192 bool patchValue(const vector<T>& values) const {
193 auto* patchedPiece = layout_.getMappedPiece<DataPieceVector<T>>(pieceIndex_);
194 return patchedPiece != nullptr && (patchedPiece->stage(values), true);
195 }
196
199 bool isAvailable() const override {
200 size_t count = 0;
201 return layout_.getVarData<T>(offset_, count) != nullptr;
202 }
203
208 void print(ostream& out, const string& indent) const override;
213 void printCompact(ostream& out, const string& indent) const override;
214
219 bool isSame(const DataPiece* rhs) const override;
220
224 void serialize(JsonWrapper& jsonWrapper, const JsonFormatProfileSpec& profile) override;
225
228 bool stageCurrentValue() override {
229 return get(stagedValues());
230 }
231
235 unique_ptr<DataPiece> clone() const override {
236 auto other = std::make_unique<DataPieceVector<T>>(getLabel());
237 other->tags_ = tags_;
238 other->required_ = required_;
239 other->defaultValues_ = defaultValues_;
240 return other;
241 }
242
243 protected:
244 bool copyFrom(const DataPiece* original) override {
245 const DataPieceVector<T>* originalVect = reinterpret_cast<const DataPieceVector<T>*>(original);
246 return originalVect->get(stagedValues_);
247 }
248
249 private:
250 vector<T> stagedValues_; // values to write to disk
251 vector<T> defaultValues_;
252};
