267class VRS_API BorrowedData {
268 public:
269 BorrowedData(const void* data, size_t size, std::shared_ptr<void> owner)
270 : segments_{{data, size}}, owner_{std::move(owner)}, size_{size} {}
271
272 BorrowedData(const DataSlice* dataSlices, size_t dataSliceCount, std::shared_ptr<void> owner)
273 : owner_{std::move(owner)} {
274 segments_.reserve(dataSliceCount);
275 for (size_t i = 0; i < dataSliceCount; ++i) {
276 segments_.push_back(dataSlices[i]);
277 size_ += dataSlices[i].size;
278 }
279 }
280
281 const std::vector<DataSlice>& getSegments() const {
282 return segments_;
283 }
284
285 size_t getDataSize() const {
286 return size_;
287 }
288
289 private:
290 std::vector<DataSlice> segments_;
291 std::shared_ptr<void> owner_;
292 size_t size_{0};
293};
